cmake_minimum_required(VERSION 3.22)
project(lma_qwen_tts_host LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Native Windows MSVC only; no virtualization")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(Threads REQUIRED)
add_subdirectory("${QWEN_SOURCE}/ggml" ggml EXCLUDE_FROM_ALL)
file(READ "${ROOT}/third_party/qwen-tts-source-lock.json" source_lock)
string(JSON source_count LENGTH "${source_lock}" compiled_sources)
math(EXPR last_source "${source_count} - 1")
set(qwen_sources)
foreach(index RANGE 0 ${last_source})
  string(JSON relative GET "${source_lock}" compiled_sources ${index})
  list(APPEND qwen_sources "${QWEN_SOURCE}/${relative}")
endforeach()
add_library(lma_qwen_core STATIC ${qwen_sources})
target_include_directories(lma_qwen_core PUBLIC "${QWEN_SOURCE}/src" "${QWEN_SOURCE}/ggml/include")
target_compile_options(lma_qwen_core PRIVATE /utf-8 /bigobj)
target_compile_definitions(lma_qwen_core PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_link_libraries(lma_qwen_core PUBLIC ggml ggml-base ggml-cpu Threads::Threads psapi)
add_executable(lma_qwen_tts_host "${ROOT}/native/qwen_tts/worker.cpp")
target_include_directories(lma_qwen_tts_host PRIVATE "${ROOT}/research/qwen_tts_host/compat" "${ROOT}/native/qwen_tts" "${ROOT}/build/native-src/sd-cpp/thirdparty")
target_compile_options(lma_qwen_tts_host PRIVATE /utf-8 /bigobj)
target_compile_definitions(lma_qwen_tts_host PRIVATE LMA_QWEN_REVISION="${LMA_QWEN_REVISION}" _CRT_SECURE_NO_WARNINGS)
target_link_libraries(lma_qwen_tts_host PRIVATE lma_qwen_core)
