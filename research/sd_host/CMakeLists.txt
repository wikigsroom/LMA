cmake_minimum_required(VERSION 3.22)
project(lma_sd_host LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Use native Windows MSVC, without virtualization")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
set(SD_CUDA OFF CACHE BOOL "" FORCE)
set(SD_VULKAN OFF CACHE BOOL "" FORCE)
set(SD_OPENCL OFF CACHE BOOL "" FORCE)
set(SD_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/build/native-src/sd-cpp" sd-source)
get_target_property(sd_sources stable-diffusion SOURCES)
list(FILTER sd_sources EXCLUDE REGEX "[/\\]tokenizers[/\\]vocab[/\\]vocab\\.cpp$")
set_property(TARGET stable-diffusion PROPERTY SOURCES ${sd_sources})
target_sources(stable-diffusion PRIVATE "${ROOT}/native/vocab_external.cpp")
target_include_directories(stable-diffusion PRIVATE "${ROOT}/build/native-src/sd-cpp/src/tokenizers/vocab")
target_compile_options(stable-diffusion PRIVATE /utf-8 /bigobj)
add_executable(lma_sd_host "${ROOT}/native/sd_worker.cpp")
target_compile_features(lma_sd_host PRIVATE cxx_std_17)
target_include_directories(lma_sd_host PRIVATE "${ROOT}/research/sd_host/compat")
target_include_directories(lma_sd_host PRIVATE "${ROOT}/build/native-src/sd-cpp/thirdparty/utf8proc")
target_compile_definitions(lma_sd_host PRIVATE LMA_SD_REVISION="${LMA_SD_REVISION}")
target_compile_options(lma_sd_host PRIVATE /utf-8 /bigobj)
target_link_libraries(lma_sd_host PRIVATE stable-diffusion)
add_executable(lma_sd_tokenizer_host tokenizer_driver.cpp)
target_include_directories(lma_sd_tokenizer_host PRIVATE "${ROOT}/native" "${ROOT}/build/native-src/sd-cpp/src" "${ROOT}/build/native-src/sd-cpp/src/tokenizers" "${ROOT}/build/native-src/sd-cpp/src/tokenizers/vocab" "${ROOT}/build/native-src/sd-cpp/thirdparty/utf8proc")
target_compile_options(lma_sd_tokenizer_host PRIVATE /utf-8 /bigobj)
target_link_libraries(lma_sd_tokenizer_host PRIVATE stable-diffusion)
