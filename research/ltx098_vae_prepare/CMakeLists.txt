cmake_minimum_required(VERSION 3.22)
project(lma_ltx098_vae_prepare LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
# Numeric bytes avoid MSVC's individual string literal limit. Metadata only.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/contract.json" CONTRACT_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," CONTRACT_BYTES "${CONTRACT_HEX}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/contract.hpp" "#pragma once\nstatic constexpr unsigned char vae_contract_bytes[]={${CONTRACT_BYTES}};\n")
add_executable(lma_ltx098_vae_prepare prepare.cpp)
target_include_directories(lma_ltx098_vae_prepare PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(lma_ltx098_vae_prepare SYSTEM PRIVATE "${LMA_JSON_SOURCE}")
if(MSVC)
    target_compile_options(lma_ltx098_vae_prepare PRIVATE /W4 /WX /fp:precise /EHsc)
else()
    target_compile_options(lma_ltx098_vae_prepare PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(ANDROID)
    target_link_options(lma_ltx098_vae_prepare PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    set_target_properties(lma_ltx098_vae_prepare PROPERTIES OUTPUT_NAME "liblma-ltx098-vae-prepare" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
endif()
