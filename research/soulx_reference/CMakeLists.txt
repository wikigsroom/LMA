cmake_minimum_required(VERSION 3.22)
project(lma_soulx_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../soulx" soulx-dsp)
add_library(lma_ort SHARED IMPORTED)
set_target_properties(lma_ort PROPERTIES IMPORTED_LOCATION "${LMA_ORT_LIBRARY}")
add_library(lma_soulx_reference STATIC reference.cpp)
target_include_directories(lma_soulx_reference PRIVATE "${LMA_ORT_HEADERS}")
target_link_libraries(lma_soulx_reference PUBLIC lma_soulx_dsp lma_ort)
target_compile_options(lma_soulx_reference PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
add_executable(lma_soulx_reference_cli driver.cpp ../soulx_ort/android_log.cpp)
target_include_directories(lma_soulx_reference_cli PRIVATE "${LMA_JSON_SOURCE}")
target_link_libraries(lma_soulx_reference_cli PRIVATE lma_soulx_reference)
if(ANDROID)
    target_link_options(lma_soulx_reference_cli PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384
        -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
    set_target_properties(lma_soulx_reference_cli PROPERTIES OUTPUT_NAME "liblma-soulx-reference" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
endif()
