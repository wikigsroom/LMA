cmake_minimum_required(VERSION 3.22)
project(lma_ltx098_vae LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(lma_ort SHARED IMPORTED)
set_target_properties(lma_ort PROPERTIES IMPORTED_LOCATION "${LMA_ORT_LIBRARY}")
if(WIN32)
    set_target_properties(lma_ort PROPERTIES IMPORTED_IMPLIB "${LMA_ORT_IMPORT_LIBRARY}")
endif()
add_executable(lma_ltx098_vae_cli driver.cpp ../soulx_ort/android_log.cpp)
target_include_directories(lma_ltx098_vae_cli SYSTEM PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}")
target_link_libraries(lma_ltx098_vae_cli PRIVATE lma_ort)
if(MSVC)
    target_compile_options(lma_ltx098_vae_cli PRIVATE /W4 /fp:precise /EHsc)
else()
    target_compile_options(lma_ltx098_vae_cli PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(ANDROID)
    target_link_options(lma_ltx098_vae_cli PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384
        -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
    set_target_properties(lma_ltx098_vae_cli PROPERTIES OUTPUT_NAME "liblma-ltx098-vae" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
endif()
