cmake_minimum_required(VERSION 3.22)
project(lma_sd_vulkan_host LANGUAGES C CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
set(SD_CUDA OFF CACHE BOOL "" FORCE)
set(SD_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(SD_OPENCL OFF CACHE BOOL "" FORCE)
set(SD_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
find_package(SPIRV-Headers CONFIG REQUIRED)
add_subdirectory("${ROOT}/build/native-src/sd-cpp" sd-source)
target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
get_target_property(sd_sources stable-diffusion SOURCES)
list(FILTER sd_sources EXCLUDE REGEX "[/\\]tokenizers[/\\]vocab[/\\]vocab\\.cpp$")
set_property(TARGET stable-diffusion PROPERTY SOURCES ${sd_sources})
target_sources(stable-diffusion PRIVATE "${ROOT}/native/vocab_external.cpp")
target_include_directories(stable-diffusion PRIVATE "${ROOT}/build/native-src/sd-cpp/src/tokenizers/vocab")
target_compile_options(stable-diffusion PRIVATE /utf-8 /bigobj)
add_executable(lma_sd_vulkan_host "${ROOT}/native/sd_worker.cpp")
target_compile_features(lma_sd_vulkan_host PRIVATE cxx_std_17)
target_include_directories(lma_sd_vulkan_host PRIVATE "${ROOT}/research/sd_host/compat" "${ROOT}/build/native-src/sd-cpp/thirdparty/utf8proc")
target_compile_definitions(lma_sd_vulkan_host PRIVATE LMA_SD_REVISION="${LMA_SD_REVISION}" LMA_SD_VULKAN=1)
target_compile_options(lma_sd_vulkan_host PRIVATE /utf-8 /bigobj)
target_link_libraries(lma_sd_vulkan_host PRIVATE stable-diffusion)
