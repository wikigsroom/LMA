cmake_minimum_required(VERSION 3.22)
project(lma_soulx_language LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    add_library(lma_llama SHARED IMPORTED)
    add_library(lma_ggml SHARED IMPORTED)
    set_target_properties(lma_llama PROPERTIES IMPORTED_LOCATION "${LMA_LLAMA_LIBRARY}")
    set_target_properties(lma_ggml PROPERTIES IMPORTED_LOCATION "${LMA_GGML_LIBRARY}")
    set(language_libs lma_llama lma_ggml dl)
else()
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
    set(GGML_RPC OFF CACHE BOOL "" FORCE)
    set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
    add_subdirectory("${LMA_LLAMA_SOURCE}" llama-source)
    set(language_libs llama)
endif()
add_library(lma_soulx_language STATIC language.cpp)
target_include_directories(lma_soulx_language PRIVATE "${LMA_LLAMA_SOURCE}/include" "${LMA_LLAMA_SOURCE}/ggml/include")
target_link_libraries(lma_soulx_language PUBLIC ${language_libs})
if(NOT MSVC)
    target_compile_options(lma_soulx_language PRIVATE -Wall -Wextra -Werror -Wno-parentheses -ffp-contract=off)
endif()
add_executable(lma_soulx_language_cli driver.cpp)
target_include_directories(lma_soulx_language_cli PRIVATE "${LMA_JSON_SOURCE}")
target_link_libraries(lma_soulx_language_cli PRIVATE lma_soulx_language)
if(NOT ANDROID)
    add_executable(lma_soulx_quantize quantize.cpp)
    target_link_libraries(lma_soulx_quantize PRIVATE llama)
    add_executable(lma_soulx_policy policy_driver.cpp)
    target_include_directories(lma_soulx_policy PRIVATE "${LMA_JSON_SOURCE}")
endif()
if(ANDROID)
    target_link_options(lma_soulx_language_cli PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    set_target_properties(lma_soulx_language_cli PROPERTIES OUTPUT_NAME "liblma-soulx-language" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
endif()
