cmake_minimum_required(VERSION 3.22)
project(lma_ltx098_text LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    add_library(lma_llama SHARED IMPORTED)
    add_library(lma_ggml SHARED IMPORTED)
    add_library(lma_ggml_base SHARED IMPORTED)
    set_target_properties(lma_llama PROPERTIES IMPORTED_LOCATION "${LMA_LLAMA_LIBRARY}")
    set_target_properties(lma_ggml PROPERTIES IMPORTED_LOCATION "${LMA_GGML_LIBRARY}")
    set_target_properties(lma_ggml_base PROPERTIES IMPORTED_LOCATION "${LMA_GGML_BASE_LIBRARY}")
    set(text_libs lma_llama lma_ggml lma_ggml_base dl)
else()
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    foreach(feature COMMON EXAMPLES TOOLS TESTS SERVER MTMD)
        set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    foreach(feature BACKEND_DL NATIVE OPENMP RPC LLAMAFILE)
        set(GGML_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory("${LMA_LLAMA_SOURCE}" llama-source)
    set(text_libs llama)
endif()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_DISABLE_EMBEDDED_DATA ON CACHE BOOL "" FORCE)
set(SPM_ENABLE_MSVC_MT_BUILD OFF CACHE BOOL "" FORCE)
add_subdirectory("${LMA_SENTENCEPIECE_SOURCE}" lma-sentencepiece EXCLUDE_FROM_ALL)
add_library(lma_ltx098_text STATIC encoder.cpp graph.cpp)
target_include_directories(lma_ltx098_text PRIVATE "${LMA_LLAMA_SOURCE}/include" "${LMA_LLAMA_SOURCE}/ggml/include" "${LMA_SENTENCEPIECE_SOURCE}/src")
target_include_directories(lma_ltx098_text SYSTEM PRIVATE "${LMA_JSON_SOURCE}")
target_link_libraries(lma_ltx098_text PUBLIC ${text_libs} sentencepiece-static)
if(NOT MSVC)
    target_compile_options(lma_ltx098_text PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
add_executable(lma_ltx098_text_cli driver.cpp)
target_include_directories(lma_ltx098_text_cli PRIVATE "${LMA_JSON_SOURCE}")
target_link_libraries(lma_ltx098_text_cli PRIVATE lma_ltx098_text)
if(ANDROID)
    target_sources(lma_ltx098_text_cli PRIVATE ../soulx_ort/android_log.cpp)
    target_link_options(lma_ltx098_text_cli PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    target_link_options(lma_ltx098_text_cli PRIVATE -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
    set_target_properties(lma_ltx098_text_cli PROPERTIES OUTPUT_NAME "liblma-ltx098-text" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
endif()
