cmake_minimum_required(VERSION 3.22)
project(lma_ltx098_prepare LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/ltx098-source-lock.json" LMA_LTX_LOCK)
foreach(FIELD checkpoint_transformer_config checkpoint_transformer_tensors checkpoint_transformer_parameters checkpoint_transformer_layout_sha256)
    string(JSON VALUE_${FIELD} GET "${LMA_LTX_LOCK}" ${FIELD})
endforeach()
set(LMA_LTX_CONTRACT "{\"checkpoint_transformer_config\":${VALUE_checkpoint_transformer_config},\"checkpoint_transformer_tensors\":${VALUE_checkpoint_transformer_tensors},\"checkpoint_transformer_parameters\":${VALUE_checkpoint_transformer_parameters},\"checkpoint_transformer_layout_sha256\":\"${VALUE_checkpoint_transformer_layout_sha256}\"}")
configure_file(contract.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/contract.hpp" @ONLY)
add_executable(lma_ltx098_prepare prepare.cpp)
target_include_directories(lma_ltx098_prepare PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(lma_ltx098_prepare SYSTEM PRIVATE "${LMA_JSON_SOURCE}")
if(MSVC)
    target_compile_options(lma_ltx098_prepare PRIVATE /W4 /WX /fp:precise /EHsc)
else()
    target_compile_options(lma_ltx098_prepare PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(ANDROID)
    target_link_options(lma_ltx098_prepare PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    set_target_properties(lma_ltx098_prepare PROPERTIES OUTPUT_NAME "liblma-ltx098-prepare" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
endif()
