cmake_minimum_required(VERSION 3.22)
project(lma_openmoss_host LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Native Windows MSVC only; no virtualization")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_subdirectory("${OPENMOSS_SOURCE}/third_party/llama.cpp" llama EXCLUDE_FROM_ALL)
add_library(lma_openmoss_core STATIC)
foreach(name tokenizer frame_decoder local_transformer dit dac_vae soundeffect)
  target_sources(lma_openmoss_core PRIVATE "${OPENMOSS_SOURCE}/src/${name}.cpp")
endforeach()
foreach(path LMA_GUARDED_PIPELINE LMA_MAPPED_MODEL LMA_GUARDED_DELAY LMA_GUARDED_CODEC)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Missing reviewed generated source: ${path}")
  endif()
endforeach()
target_sources(lma_openmoss_core PRIVATE "${LMA_GUARDED_PIPELINE}" "${LMA_MAPPED_MODEL}" "${LMA_GUARDED_DELAY}" "${LMA_GUARDED_CODEC}")
target_include_directories(lma_openmoss_core PRIVATE "${ROOT}/native/openmoss" "${OPENMOSS_SOURCE}/src" "${OPENMOSS_SOURCE}/third_party/llama.cpp/ggml/src")
target_include_directories(lma_openmoss_core PUBLIC "${OPENMOSS_SOURCE}/include")
target_compile_options(lma_openmoss_core PRIVATE /utf-8 /bigobj)
target_compile_definitions(lma_openmoss_core PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_link_libraries(lma_openmoss_core PUBLIC llama ggml ggml-base ggml-cpu)
add_executable(lma_openmoss_host "${ROOT}/native/openmoss/worker.cpp")
target_include_directories(lma_openmoss_host PRIVATE "${ROOT}/research/qwen_tts_host/compat" "${ROOT}/native/openmoss" "${OPENMOSS_SOURCE}/src" "${ROOT}/build/native-src/sd-cpp/thirdparty" "${OPENMOSS_SOURCE}/third_party/llama.cpp/ggml/src")
target_compile_options(lma_openmoss_host PRIVATE /utf-8 /bigobj)
target_compile_definitions(lma_openmoss_host PRIVATE LMA_OPENMOSS_REVISION="${LMA_OPENMOSS_REVISION}" _CRT_SECURE_NO_WARNINGS)
target_link_libraries(lma_openmoss_host PRIVATE lma_openmoss_core)
add_executable(lma_openmoss_mapping_test "${ROOT}/tests/native/openmoss_mapping_test.cpp")
target_include_directories(lma_openmoss_mapping_test PRIVATE "${ROOT}/native/openmoss" "${OPENMOSS_SOURCE}/third_party/llama.cpp/ggml/src")
target_compile_definitions(lma_openmoss_mapping_test PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_compile_options(lma_openmoss_mapping_test PRIVATE /utf-8)
target_link_libraries(lma_openmoss_mapping_test PRIVATE ggml ggml-base ggml-cpu)
add_executable(lma_openmoss_codec_memory_test "${ROOT}/tests/native/openmoss_codec_memory_test.cpp")
target_include_directories(lma_openmoss_codec_memory_test PRIVATE "${ROOT}/native/openmoss")
target_compile_options(lma_openmoss_codec_memory_test PRIVATE /utf-8)
target_link_libraries(lma_openmoss_codec_memory_test PRIVATE ggml ggml-base ggml-cpu)
