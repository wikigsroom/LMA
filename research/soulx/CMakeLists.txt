cmake_minimum_required(VERSION 3.22)
project(lma_soulx_dsp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Explicit local, reviewed sources. No FetchContent/downloads during compilation.
if(NOT EXISTS "${LMA_KISSFFT_SOURCE}/kiss_fftr.c")
    message(FATAL_ERROR "Run tools/fetch_soulx_dsp.py first")
endif()
add_library(lma_soulx_dsp STATIC dsp.cpp "${LMA_KISSFFT_SOURCE}/kiss_fft.c" "${LMA_KISSFFT_SOURCE}/kiss_fftr.c")
target_include_directories(lma_soulx_dsp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${LMA_KISSFFT_SOURCE}")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(lma_soulx_dsp PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
