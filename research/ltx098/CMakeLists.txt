cmake_minimum_required(VERSION 3.22)
project(lma_ltx098 LANGUAGES C CXX)
option(LMA_LTX_VULKAN "Compile the explicit Vulkan backend" OFF)
if(NOT EXISTS "${LMA_GGML_SOURCE}/include/ggml.h")
  message(FATAL_ERROR "Set LMA_GGML_SOURCE to the pinned ggml checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ${LMA_LTX_VULKAN} CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
if(LMA_LTX_VULKAN)
  find_package(SPIRV-Headers CONFIG REQUIRED)
endif()
add_subdirectory("${LMA_GGML_SOURCE}" ggml)
if(LMA_LTX_VULKAN)
  # This pinned GGML revision finds this package but does not propagate its
  # imported include path when building outside a globally installed Vulkan SDK.
  target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif()
add_executable(lma_ltx098_probe driver.cpp transformer.cpp)
target_include_directories(lma_ltx098_probe SYSTEM PRIVATE "${LMA_JSON_INCLUDE}")
target_link_libraries(lma_ltx098_probe PRIVATE ggml)
target_compile_features(lma_ltx098_probe PRIVATE cxx_std_17)
if(LMA_LTX_VULKAN)
  target_compile_definitions(lma_ltx098_probe PRIVATE LMA_LTX_VULKAN=1)
endif()
if(MSVC)
  target_compile_options(lma_ltx098_probe PRIVATE /W4 /fp:precise)
else()
  target_compile_options(lma_ltx098_probe PRIVATE -Wall -Wextra -Werror -fno-fast-math)
endif()
if(ANDROID)
  if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Only Android ARM64 is reviewed")
  endif()
  target_link_options(lma_ltx098_probe PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()
