cmake_minimum_required(VERSION 3.22)
project(lma_ltx098_tensors LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(lma_ltx098_tensors tensors.cpp)
target_include_directories(lma_ltx098_tensors SYSTEM PRIVATE "${LMA_JSON_SOURCE}")
if(MSVC)
    target_compile_options(lma_ltx098_tensors PRIVATE /W4 /fp:precise /EHsc)
else()
    target_compile_options(lma_ltx098_tensors PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(ANDROID)
    target_link_options(lma_ltx098_tensors PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    set_target_properties(lma_ltx098_tensors PROPERTIES OUTPUT_NAME "liblma-ltx098-tensors" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
endif()
