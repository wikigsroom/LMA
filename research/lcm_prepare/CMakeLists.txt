cmake_minimum_required(VERSION 3.22)
project(lma_lcm_prepare LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/contract.json" LMA_LCM_CONTRACT)
configure_file(contract.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/contract.hpp" @ONLY)
add_executable(lma_lcm_prepare prepare.cpp)
target_include_directories(lma_lcm_prepare PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(lma_lcm_prepare SYSTEM PRIVATE "${LMA_JSON_SOURCE}")
if(MSVC)
    target_compile_options(lma_lcm_prepare PRIVATE /W4 /WX /fp:strict /EHsc)
else()
    target_compile_options(lma_lcm_prepare PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endif()
if(ANDROID)
    target_link_options(lma_lcm_prepare PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    set_target_properties(lma_lcm_prepare PROPERTIES OUTPUT_NAME "liblma-lcm-prepare" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
endif()
