cmake_minimum_required(VERSION 3.22)
project(lma_native LANGUAGES C CXX)

if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "LMA native workers require an Android ARM64 NDK build")
endif()
if(NOT EXISTS "${LMA_SD_SOURCE}/include/stable-diffusion.h")
    message(FATAL_ERROR "Set LMA_SD_SOURCE to the pinned stable-diffusion.cpp checkout")
endif()
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
add_subdirectory("${LMA_SD_SOURCE}" sd-source)

# Upstream embeds model-specific vocabularies. Load pinned user-selected data instead.
get_target_property(sd_sources stable-diffusion SOURCES)
list(FILTER sd_sources EXCLUDE REGEX "[/\\]tokenizers[/\\]vocab[/\\]vocab\\.cpp$")
set_property(TARGET stable-diffusion PROPERTY SOURCES ${sd_sources})
target_sources(stable-diffusion PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vocab_external.cpp")
target_include_directories(stable-diffusion PRIVATE "${LMA_SD_SOURCE}/src/tokenizers/vocab")

add_executable(lma_sd_worker sd_worker.cpp)
target_link_libraries(lma_sd_worker PRIVATE stable-diffusion)
target_include_directories(lma_sd_worker PRIVATE "${LMA_SD_SOURCE}/thirdparty/utf8proc")
target_compile_features(lma_sd_worker PRIVATE cxx_std_17)
target_compile_definitions(lma_sd_worker PRIVATE LMA_SD_REVISION="${LMA_SD_REVISION}")
target_link_options(lma_sd_worker PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
set_target_properties(lma_sd_worker PROPERTIES OUTPUT_NAME "liblma-sd" SUFFIX ".so")
