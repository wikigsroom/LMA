cmake_minimum_required(VERSION 3.22)
project(lma_moss_nano LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Use the Android ARM64 NDK toolchain")
endif()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_DISABLE_EMBEDDED_DATA ON CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${LMA_SENTENCEPIECE_SOURCE}" sentencepiece EXCLUDE_FROM_ALL)
add_library(lma_ort SHARED IMPORTED)
set_target_properties(lma_ort PROPERTIES IMPORTED_LOCATION "${LMA_ORT_LIBRARY}")
add_executable(lma_moss_nano_worker worker.cpp)
target_include_directories(lma_moss_nano_worker PRIVATE "${LMA_SENTENCEPIECE_SOURCE}/src" "${LMA_ORT_SOURCE}/include/onnxruntime/core/session" "${LMA_SD_SOURCE}/thirdparty")
target_link_libraries(lma_moss_nano_worker PRIVATE sentencepiece-static lma_ort)
target_compile_features(lma_moss_nano_worker PRIVATE cxx_std_17)
target_compile_definitions(lma_moss_nano_worker PRIVATE LMA_MOSS_REVISION="${LMA_MOSS_REVISION}")
target_link_options(lma_moss_nano_worker PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384
  -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
set_target_properties(lma_moss_nano_worker PROPERTIES OUTPUT_NAME "liblma-moss-nano" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
