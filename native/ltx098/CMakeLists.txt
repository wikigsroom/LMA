cmake_minimum_required(VERSION 3.22)
project(lma_ltx098_app LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
if(LMA_STAGE STREQUAL "sample")
    add_subdirectory("${root}/research/ltx098" component)
    add_executable(lma_ltx098_app_sample sample.cpp parent_guard.cpp)
    target_include_directories(lma_ltx098_app_sample PRIVATE "${root}/research/ltx098")
    target_link_libraries(lma_ltx098_app_sample PRIVATE lma_ltx098_core)
    if(NOT ANDROID)
        add_executable(lma_ltx098_multiscale_contract program_contract.cpp)
        target_include_directories(lma_ltx098_multiscale_contract PRIVATE "${root}/research/ltx098" "${LMA_JSON_SOURCE}")
        target_link_libraries(lma_ltx098_multiscale_contract PRIVATE lma_ltx098_core)
        set_target_properties(lma_ltx098_multiscale_contract PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    endif()
    set(worker lma_ltx098_app_sample)
elseif(LMA_STAGE STREQUAL "text")
    add_subdirectory("${root}/research/ltx098_text" component)
    set(worker lma_ltx098_text_cli)
elseif(LMA_STAGE STREQUAL "vae")
    add_subdirectory("${root}/research/ltx098_vae" component)
    set(worker lma_ltx098_vae_cli)
elseif(LMA_STAGE STREQUAL "tensors")
    add_subdirectory("${root}/research/ltx098_pipeline" component)
    set(worker lma_ltx098_tensors)
else()
    message(FATAL_ERROR "Select a signed App stage")
endif()
if(NOT LMA_STAGE STREQUAL "sample")
    target_sources(${worker} PRIVATE parent_guard.cpp)
endif()
if(MSVC)
    target_compile_options(${worker} PRIVATE /fp:precise /EHsc)
else()
    target_compile_options(${worker} PRIVATE -ffp-contract=off)
endif()
if(ANDROID)
    target_link_options(${worker} PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    set_target_properties(${worker} PROPERTIES OUTPUT_NAME "liblma-ltx098-app-${LMA_STAGE}" SUFFIX ".so" SKIP_BUILD_RPATH TRUE RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
else()
    set_target_properties(${worker} PROPERTIES OUTPUT_NAME "lma-ltx098-app-${LMA_STAGE}" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
