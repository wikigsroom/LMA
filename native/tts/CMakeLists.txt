cmake_minimum_required(VERSION 3.22)
project(lma_tts LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Use the Android ARM64 NDK toolchain")
endif()
add_library(sherpa_c_api SHARED IMPORTED)
set_target_properties(sherpa_c_api PROPERTIES IMPORTED_LOCATION "${LMA_SHERPA_LIBS}/libsherpa-onnx-c-api.so")
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${LMA_SHERPA_LIBS}/libonnxruntime.so")
add_executable(lma_tts_worker worker.cpp)
target_include_directories(lma_tts_worker PRIVATE "${LMA_SHERPA_SOURCE}" "${LMA_SD_SOURCE}/thirdparty")
target_link_libraries(lma_tts_worker PRIVATE sherpa_c_api onnxruntime)
target_compile_features(lma_tts_worker PRIVATE cxx_std_17)
target_compile_definitions(lma_tts_worker PRIVATE LMA_SHERPA_REVISION="${LMA_SHERPA_REVISION}")
target_link_options(lma_tts_worker PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384
  -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
set_target_properties(lma_tts_worker PROPERTIES OUTPUT_NAME "liblma-tts" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
