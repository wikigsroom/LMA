cmake_minimum_required(VERSION 3.22)
project(lma_rf_shape LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(lma_ort SHARED IMPORTED)
set_target_properties(lma_ort PROPERTIES IMPORTED_LOCATION "${LMA_ORT_LIBRARY}")
if(WIN32)
 set_target_properties(lma_ort PROPERTIES IMPORTED_IMPLIB "${LMA_ORT_IMPORT_LIBRARY}")
endif()
add_executable(lma_rf_shape worker.cpp ../../research/soulx_ort/android_log.cpp)
target_include_directories(lma_rf_shape PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")
target_compile_definitions(lma_rf_shape PRIVATE LMA_TRIPO_REVISION="fc5c40990181e2a756c4e0b1c2f4d6b5202faf8c" LMA_PART_REVISION="3d773bf02fad51c7ab31a5615573fec93b287b30")
target_link_libraries(lma_rf_shape PRIVATE lma_ort)
if(ANDROID)
 target_link_options(lma_rf_shape PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384
  -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
 set_target_properties(lma_rf_shape PROPERTIES OUTPUT_NAME "liblma-rf-shape" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
else()
 set_target_properties(lma_rf_shape PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(lma_shape_contract ../../tests/expansion-native/shape.cpp)
 target_include_directories(lma_shape_contract PRIVATE "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")
 set_target_properties(lma_shape_contract PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
