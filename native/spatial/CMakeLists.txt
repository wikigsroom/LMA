cmake_minimum_required(VERSION 3.22)
project(lma_spatial LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 add_compile_options(/utf-8)
endif()
file(GLOB LMA_PROGRAM_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/../rf_shape/*.hpp"
 "${CMAKE_CURRENT_SOURCE_DIR}/../worker_common.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/../runtime_metrics.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/../moss_nano/onnx_external.hpp")
# Explicit dependencies also work with localized MSVC /showIncludes output.
set_source_files_properties(worker.cpp ../../tests/expansion-native/spatial.cpp ../../tests/expansion-native/sparse_graph.cpp PROPERTIES OBJECT_DEPENDS "${LMA_PROGRAM_HEADERS}")
add_library(lma_ort SHARED IMPORTED)
set_target_properties(lma_ort PROPERTIES IMPORTED_LOCATION "${LMA_ORT_LIBRARY}")
if(WIN32)
 set_target_properties(lma_ort PROPERTIES IMPORTED_IMPLIB "${LMA_ORT_IMPORT_LIBRARY}")
endif()
add_executable(lma_spatial worker.cpp ../../research/soulx_ort/android_log.cpp)
target_include_directories(lma_spatial PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")
target_link_libraries(lma_spatial PRIVATE lma_ort)
if(ANDROID)
 target_link_options(lma_spatial PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384
  -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
 set_target_properties(lma_spatial PROPERTIES OUTPUT_NAME "liblma-spatial" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
else()
 set_target_properties(lma_spatial PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(lma_spatial_contract ../../tests/expansion-native/spatial.cpp)
 target_include_directories(lma_spatial_contract PRIVATE "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")
 set_target_properties(lma_spatial_contract PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(lma_sparse_graph ../../tests/expansion-native/sparse_graph.cpp)
 target_include_directories(lma_sparse_graph PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}")
 target_link_libraries(lma_sparse_graph PRIVATE lma_ort)
 set_target_properties(lma_sparse_graph PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
