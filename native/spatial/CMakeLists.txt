cmake_minimum_required(VERSION 3.22)
project(lma_spatial LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 add_compile_options(/utf-8)
endif()
file(GLOB LMA_PROGRAM_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/../rf_shape/*.hpp"
 "${CMAKE_CURRENT_SOURCE_DIR}/../worker_common.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/../runtime_metrics.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/../moss_nano/onnx_external.hpp")
# Explicit dependencies also work with localized MSVC /showIncludes output.
set_source_files_properties(worker.cpp ../../tests/expansion-native/spatial.cpp ../../tests/expansion-native/sparse_graph.cpp ../../tests/expansion-native/material.cpp PROPERTIES OBJECT_DEPENDS "${LMA_PROGRAM_HEADERS}")
add_library(lma_ort SHARED IMPORTED)
set_target_properties(lma_ort PROPERTIES IMPORTED_LOCATION "${LMA_ORT_LIBRARY}")
if(WIN32)
 set_target_properties(lma_ort PROPERTIES IMPORTED_IMPLIB "${LMA_ORT_IMPORT_LIBRARY}")
endif()
add_executable(lma_spatial worker.cpp ../../research/soulx_ort/android_log.cpp)
add_library(lma_xatlas STATIC "${LMA_XATLAS_SOURCE}/xatlas.cpp")
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_DISABLE_EMBEDDED_DATA ON CACHE BOOL "" FORCE)
set(SPM_ENABLE_MSVC_MT_BUILD OFF CACHE BOOL "" FORCE)
add_subdirectory("${LMA_SENTENCEPIECE_SOURCE}" lma-sentencepiece EXCLUDE_FROM_ALL)
set(SD_WEBP OFF)
set(SD_WEBM OFF)
add_subdirectory("${LMA_TOKENIZER_SOURCE}/thirdparty" lma-tokenizer-deps EXCLUDE_FROM_ALL)
add_library(lma_text_tokens STATIC
 "${LMA_TOKENIZER_SOURCE}/src/tokenizers/hf_tokenizer.cpp"
 "${LMA_TOKENIZER_SOURCE}/src/tokenizers/tokenizer.cpp"
 "${LMA_TOKENIZER_SOURCE}/src/core/regex.cpp")
target_include_directories(lma_text_tokens PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tokenizer_shim")
target_include_directories(lma_text_tokens PUBLIC "${LMA_SENTENCEPIECE_SOURCE}/src" "${LMA_TOKENIZER_SOURCE}/src" "${LMA_TOKENIZER_SOURCE}/thirdparty")
target_link_libraries(lma_text_tokens PRIVATE onig sd-utf8proc)
target_include_directories(lma_xatlas PUBLIC "${LMA_XATLAS_SOURCE}")
target_include_directories(lma_spatial PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")
target_include_directories(lma_spatial PRIVATE "${LMA_FDG_SOURCE}" "${LMA_EIGEN_SOURCE}")
target_compile_definitions(lma_spatial PRIVATE EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
target_link_libraries(lma_spatial PRIVATE lma_ort lma_xatlas lma_text_tokens sentencepiece-static)
if(ANDROID)
 target_link_options(lma_spatial PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384
  -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
 set_target_properties(lma_spatial PROPERTIES OUTPUT_NAME "liblma-spatial" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
else()
 set_target_properties(lma_spatial PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(lma_spatial_contract ../../tests/expansion-native/spatial.cpp)
 target_include_directories(lma_spatial_contract PRIVATE "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")
 set_target_properties(lma_spatial_contract PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(lma_sparse_graph ../../tests/expansion-native/sparse_graph.cpp)
 target_include_directories(lma_sparse_graph PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}")
 target_include_directories(lma_sparse_graph PRIVATE "${LMA_FDG_SOURCE}" "${LMA_EIGEN_SOURCE}")
 target_compile_definitions(lma_sparse_graph PRIVATE EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
 target_link_libraries(lma_sparse_graph PRIVATE lma_ort)
 set_target_properties(lma_sparse_graph PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(lma_material_contract ../../tests/expansion-native/material.cpp)
 target_include_directories(lma_material_contract PRIVATE "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")
 target_link_libraries(lma_material_contract PRIVATE lma_xatlas)
 set_target_properties(lma_material_contract PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(lma_text_contract ../../tests/expansion-native/text_tokens.cpp)
 target_include_directories(lma_text_contract PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}")
 target_link_libraries(lma_text_contract PRIVATE lma_ort lma_text_tokens sentencepiece-static)
 set_target_properties(lma_text_contract PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(lma_ultratex_contract ../../tests/expansion-native/ultratex.cpp)
 target_include_directories(lma_ultratex_contract PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")

endif()
