function(lma_mesh_progress marker stage expected)
  string(LENGTH "${lma_cli}" before)
  string(REPLACE "${marker}" "" without "${lma_cli}")
  string(LENGTH "${without}" after)
  string(LENGTH "${marker}" size)
  math(EXPR count "(${before}-${after})/${size}")
  if(NOT count EQUAL expected)
    message(FATAL_ERROR "Pinned TRELLIS progress anchor changed: ${stage}")
  endif()
  string(REPLACE "${marker}" "lma_mesh_stage(\"${stage}\"); ${marker}" updated "${lma_cli}")
  set(lma_cli "${updated}" PARENT_SCOPE)
endfunction()
