cmake_minimum_required(VERSION 3.22)
project(lma_vision LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_TESTS LLAMA_BUILD_SERVER LLAMA_CURL LLAMA_SUBPROCESS MTMD_VIDEO GGML_BACKEND_DL GGML_NATIVE GGML_OPENMP GGML_RPC GGML_LLAMAFILE GGML_CUDA GGML_VULKAN GGML_METAL)
 set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
add_subdirectory("${LMA_LLAMA_SOURCE}" llama)
add_library(lma_ort SHARED IMPORTED)
set_target_properties(lma_ort PROPERTIES IMPORTED_LOCATION "${LMA_ORT_LIBRARY}")
if(WIN32)
 set_target_properties(lma_ort PROPERTIES IMPORTED_IMPLIB "${LMA_ORT_IMPORT_LIBRARY}")
endif()
file(GLOB LMA_VISION_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/../rf_shape/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/../worker_common.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/../runtime_metrics.hpp")
set_source_files_properties(worker.cpp ../../tests/expansion-native/vision.cpp PROPERTIES OBJECT_DEPENDS "${LMA_VISION_HEADERS}")
add_executable(lma_vision_worker worker.cpp ../../research/soulx_ort/android_log.cpp)
target_include_directories(lma_vision_worker PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")
target_link_libraries(lma_vision_worker PRIVATE lma_ort mtmd llama)
if(ANDROID)
 target_link_options(lma_vision_worker PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
 set_target_properties(lma_vision_worker PROPERTIES OUTPUT_NAME "liblma-vision" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
else()
 set_target_properties(lma_vision_worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(lma_vision_contract ../../tests/expansion-native/vision.cpp)
 target_include_directories(lma_vision_contract PRIVATE "${LMA_ORT_HEADERS}" "${LMA_JSON_SOURCE}" "${LMA_STB_SOURCE}")
 target_link_libraries(lma_vision_contract PRIVATE lma_ort mtmd llama)
 set_target_properties(lma_vision_contract PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
