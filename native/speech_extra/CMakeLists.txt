cmake_minimum_required(VERSION 3.20)
project(lma_speech_extra LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AUDIOCPP_MODEL_SET custom CACHE STRING "" FORCE)
set(AUDIOCPP_MODELS "f5_tts;omnivoice" CACHE STRING "" FORCE)
set(AUDIOCPP_DEPLOYMENT_BUILD ON CACHE BOOL "" FORCE)
set(AUDIOCPP_BUILD_NATIVE_MODEL_MANAGER OFF CACHE BOOL "" FORCE)
set(AUDIOCPP_STATIC_ESPEAK OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_NATIVE_CPU OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_CUDA OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(LMA_EMBEDDING_ONLY ON CACHE BOOL "" FORCE)
add_subdirectory("${LMA_AUDIO_SOURCE}" audio EXCLUDE_FROM_ALL)
file(READ "${LMA_AUDIO_SOURCE}/src/community_models/f5_tts/synthesize.cpp" f5_source)
set(needle "    return std::string(dialect_token(dialect))")
string(FIND "${f5_source}" "${needle}" found)
if(found LESS 0)
 message(FATAL_ERROR "Pinned F5 prompt assembly changed")
endif()
string(REPLACE "${needle}" "    if (lma_f5::active) return ref_text + chunk;\n${needle}" f5_source "${f5_source}")
set(needle "    std::vector<int32_t> ids;")
string(FIND "${f5_source}" "${needle}" found)
if(found LESS 0)
 message(FATAL_ERROR "Pinned F5 tokenizer changed")
endif()
string(REPLACE "${needle}" "    if (lma_f5::active) return lma_f5::tokenize(vocab, s);\n${needle}" f5_source "${f5_source}")
set(needle "        all_rows.insert(all_rows.end(), out.gen_mel_rows.begin(), out.gen_mel_rows.end());")
string(FIND "${f5_source}" "${needle}" found)
if(found LESS 0)
 message(FATAL_ERROR "Pinned F5 output assembly changed")
endif()
string(REPLACE "${needle}" "        if (lma_f5::active && all_rows.size() + out.gen_mel_rows.size() > lma_f5::maximum_mel_frames * 100) throw std::runtime_error(\"F5 output exceeds selected duration; split the text\");\n${needle}" f5_source "${f5_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/f5_v1_synthesize.cpp" "#include \"f5_frontend.hpp\"\n${f5_source}")
get_target_property(f5_sources engine_model_f5_tts SOURCES)
list(FILTER f5_sources EXCLUDE REGEX "synthesize\\.cpp$")
set_property(TARGET engine_model_f5_tts PROPERTY SOURCES "${f5_sources}")
target_sources(engine_model_f5_tts PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/f5_v1_synthesize.cpp")
target_include_directories(engine_model_f5_tts PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${LMA_JSON_SOURCE}" "${LMA_AUDIO_SOURCE}/src/community_models/f5_tts")
add_executable(lma-speech-extra worker.cpp "${LMA_AUDIO_SOURCE}/src/capi/audiocpp.cpp")
target_include_directories(lma-speech-extra PRIVATE "${LMA_JSON_SOURCE}")
target_link_libraries(lma-speech-extra PRIVATE engine_runtime)
if(ANDROID)
  set_target_properties(lma-speech-extra PROPERTIES OUTPUT_NAME liblma-speech-extra.so)
  target_link_options(lma-speech-extra PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,--export-dynamic)
endif()
