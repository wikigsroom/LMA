cmake_minimum_required(VERSION 3.22)
project(lma_stable_audio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_DISABLE_EMBEDDED_DATA ON CACHE BOOL "" FORCE)
add_subdirectory("${LMA_SENTENCEPIECE_SOURCE}" sentencepiece EXCLUDE_FROM_ALL)
set(LITERT_BUILD_CONFIG_DISABLE_GPU 1)
set(LITERT_BUILD_CONFIG_DISABLE_NPU 1)
configure_file("${LMA_LITERT_SDK}/litert/build_common/build_config.h.in" "${CMAKE_BINARY_DIR}/include/litert/build_common/build_config.h")
add_library(litert_runtime SHARED IMPORTED)
set_target_properties(litert_runtime PROPERTIES IMPORTED_LOCATION "${LMA_LITERT_RUNTIME}")
add_executable(lma_stable_audio_worker worker.cpp)
target_include_directories(lma_stable_audio_worker PRIVATE "${LMA_LITERT_SDK}" "${CMAKE_BINARY_DIR}/include" "${LMA_JSON_SOURCE}" "${LMA_SENTENCEPIECE_SOURCE}/src")
target_compile_definitions(lma_stable_audio_worker PRIVATE LMA_SAO_REVISION="${LMA_SAO_REVISION}")
target_link_libraries(lma_stable_audio_worker PRIVATE sentencepiece-static litert_runtime)
target_link_options(lma_stable_audio_worker PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
set_target_properties(lma_stable_audio_worker PROPERTIES OUTPUT_NAME "liblma-stable-audio" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
