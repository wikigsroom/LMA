cmake_minimum_required(VERSION 3.22)
project(lma_soulx_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../research/soulx" soulx-dsp)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../research/soulx_language" soulx-language)
add_library(lma_ort SHARED IMPORTED)
set_target_properties(lma_ort PROPERTIES IMPORTED_LOCATION "${LMA_ORT_LIBRARY}")
if(WIN32)
    set_target_properties(lma_ort PROPERTIES IMPORTED_IMPLIB "${LMA_ORT_IMPORT_LIBRARY}")
endif()
add_executable(lma_soulx_worker worker.cpp ../../research/soulx_reference/reference.cpp
    ../../research/soulx_acoustic/acoustic.cpp ../../research/soulx_ort/android_log.cpp)
target_include_directories(lma_soulx_worker PRIVATE "${LMA_ORT_HEADERS}")
target_include_directories(lma_soulx_worker SYSTEM PRIVATE "${LMA_JSON_SOURCE}")
target_link_libraries(lma_soulx_worker PRIVATE lma_soulx_dsp lma_soulx_language lma_ort)
if(MSVC)
    target_compile_options(lma_soulx_worker PRIVATE /W4 /fp:precise)
else()
    target_compile_options(lma_soulx_worker PRIVATE -Wall -Wextra -Werror -Wno-parentheses -ffp-contract=off)
endif()
if(ANDROID)
    target_link_options(lma_soulx_worker PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384
        -Wl,--export-dynamic-symbol=__android_log_print -Wl,--export-dynamic-symbol=__android_log_vprint -Wl,--export-dynamic-symbol=__android_log_write)
    set_target_properties(lma_soulx_worker PROPERTIES OUTPUT_NAME "liblma-soulx" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
else()
    set_target_properties(lma_soulx_worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    add_executable(lma_soulx_contract_test ../../tests/native/soulx_contract_test.cpp)
    target_include_directories(lma_soulx_contract_test PRIVATE "${LMA_JSON_SOURCE}")
endif()
