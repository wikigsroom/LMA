cmake_minimum_required(VERSION 3.22)
project(lma_moss_delay LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${LMA_MOSS_SOURCE}" llama EXCLUDE_FROM_ALL)
add_executable(lma_moss_delay_worker worker.cpp)
target_include_directories(lma_moss_delay_worker PRIVATE "${LMA_JSON_SOURCE}" "${LMA_MOSS_GENERATED}")
target_compile_definitions(lma_moss_delay_worker PRIVATE LMA_MOSS_REVISION="${LMA_MOSS_REVISION}")
target_compile_options(lma_moss_delay_worker PRIVATE -fexceptions)
target_link_libraries(lma_moss_delay_worker PRIVATE llama)
target_link_options(lma_moss_delay_worker PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
set_target_properties(lma_moss_delay_worker PROPERTIES OUTPUT_NAME "liblma-moss-delay" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
