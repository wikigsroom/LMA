cmake_minimum_required(VERSION 3.22)
project(lma_openmoss LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${LMA_OPENMOSS_SOURCE}/third_party/llama.cpp" llama EXCLUDE_FROM_ALL)
add_library(lma_openmoss_core STATIC)
foreach(name tokenizer delay frame_decoder local_transformer codec dit dac_vae soundeffect)
    target_sources(lma_openmoss_core PRIVATE "${LMA_OPENMOSS_SOURCE}/src/${name}.cpp")
endforeach()
target_sources(lma_openmoss_core PRIVATE "${LMA_GUARDED_PIPELINE}" "${LMA_MAPPED_MODEL}")
target_include_directories(lma_openmoss_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${LMA_OPENMOSS_SOURCE}/src" "${LMA_OPENMOSS_SOURCE}/third_party/llama.cpp/ggml/src")
target_include_directories(lma_openmoss_core PUBLIC "${LMA_OPENMOSS_SOURCE}/include")
target_link_libraries(lma_openmoss_core PUBLIC llama ggml ggml-base)
add_executable(lma_openmoss_worker worker.cpp)
target_include_directories(lma_openmoss_worker PRIVATE "${LMA_JSON_SOURCE}" "${LMA_OPENMOSS_SOURCE}/src" "${LMA_OPENMOSS_SOURCE}/third_party/llama.cpp/ggml/src")
target_compile_definitions(lma_openmoss_worker PRIVATE LMA_OPENMOSS_REVISION="${LMA_OPENMOSS_REVISION}")
target_link_libraries(lma_openmoss_worker PRIVATE lma_openmoss_core)
target_link_options(lma_openmoss_worker PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
set_target_properties(lma_openmoss_worker PROPERTIES OUTPUT_NAME "liblma-openmoss" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
