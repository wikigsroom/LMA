cmake_minimum_required(VERSION 3.22)
project(lma_moss_rq LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${LMA_MOSS_RQ_SOURCE}/third_party/ggml" ggml EXCLUDE_FROM_ALL)
add_library(lma_moss_rq_core STATIC codec_binding.cpp)
foreach(name common backend parallel profiler model_loader audio_io patchify transformer quantizer audio_tokenizer qwen3 delay_backbone sampling de_tokenizer local_transformer local_embeddings local_adapters prompt_local rt_local rt_embeddings rt_heads prompt_rt gpt2)
    target_sources(lma_moss_rq_core PRIVATE "${LMA_MOSS_RQ_SOURCE}/src/${name}.cpp")
endforeach()
target_sources(lma_moss_rq_core PRIVATE "${LMA_MOSS_RQ_GENERATED}/moss_tts_local.cpp" "${LMA_MOSS_RQ_GENERATED}/moss_tts_rt.cpp")
target_include_directories(lma_moss_rq_core PUBLIC "${LMA_MOSS_RQ_GENERATED}" "${LMA_MOSS_RQ_SOURCE}/include" "${LMA_MOSS_RQ_SOURCE}/src" PRIVATE "${LMA_MOSS_RQ_SOURCE}/third_party" "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Threads REQUIRED)
target_link_libraries(lma_moss_rq_core PUBLIC ggml Threads::Threads)
if(ANDROID)
    add_executable(lma_moss_rq_worker worker.cpp)
    target_include_directories(lma_moss_rq_worker PRIVATE "${LMA_JSON_SOURCE}")
    target_compile_definitions(lma_moss_rq_worker PRIVATE LMA_MOSS_RQ_REVISION="${LMA_MOSS_RQ_REVISION}")
    target_link_libraries(lma_moss_rq_worker PRIVATE lma_moss_rq_core)
    target_link_options(lma_moss_rq_worker PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
    set_target_properties(lma_moss_rq_worker PROPERTIES OUTPUT_NAME "liblma-moss-rq" SUFFIX ".so" SKIP_BUILD_RPATH TRUE)
else()
    add_executable(test_rq_layout "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/native/moss_rq_layout_test.cpp")
    target_link_libraries(test_rq_layout PRIVATE lma_moss_rq_core)
    add_executable(test_codec_stream "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/native/moss_codec_stream_test.cpp")
    target_link_libraries(test_codec_stream PRIVATE lma_moss_rq_core)
    foreach(name local_block rt_depth_loop)
        add_executable(test_${name} "${LMA_MOSS_RQ_SOURCE}/tests/test_${name}.cpp")
        target_link_libraries(test_${name} PRIVATE lma_moss_rq_core)
    endforeach()
endif()
